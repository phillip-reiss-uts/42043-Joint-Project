# ---- core simulation -------------------------------------------------------
# mujoco >= 3.0 for the passive viewer and MjSpec; 3.10 is what was tested.
mujoco>=3.0.0
# Fallback source of MJCF models. It is NOT needed for this project's gripper:
# the RG2 is vendored at models/onrobot_rg2/. menagerie has no UR3e either --
# that MJCF is vendored under models/ur3e/. Only a model name that models/ does
# not resolve reaches menagerie at all.
mujoco-menagerie>=0.1.0

# ---- maths -----------------------------------------------------------------
numpy>=1.24.0
# expm for the discrete observer, solve_discrete_are for the LQR design.
scipy>=1.10.0
# run_tests.py plots; nothing on the control path needs it.
matplotlib>=3.11.1

# ---- hardware only (NOT installed here, not needed for simulation) ----------

# onRobot            # the RG2 client (XML-RPC to the OnRobot URCap, port 41414):

# pyrealsense2       # D435i depth stream for RealObjectWorld

# Python bindings.
